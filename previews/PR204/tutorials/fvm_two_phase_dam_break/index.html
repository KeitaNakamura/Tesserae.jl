Finite-volume two-phase flow

The MPM dam break uses a 1.2 m by 0.6 m water column in a 3.22 m by 2.5 m tank. Here the same two-dimensional case is solved on a MAC grid, with incompressible water and air. The equations are

\[\nabla\cdot\boldsymbol u=0,\qquad \partial_t(\rho\boldsymbol u)+\nabla\cdot(\rho\boldsymbol u\otimes\boldsymbol u) =-\nabla p+\nabla\cdot\boldsymbol\tau+\rho\boldsymbol g,\qquad \boldsymbol\tau=\mu(\nabla\boldsymbol u+\nabla\boldsymbol u^\top).\]

Here $c$ is the water volume fraction. It is transported by geometric PLIC/VOF[1]. The water flux also supplies the mass flux used in the momentum balance on a staggered MAC grid[2]. The example uses impermeable, free-slip side walls and floor, and an atmospheric open top. Surface tension is omitted. Density and viscosity are volume averages of the water and air properties in each cell.

The VTK output is written to output/fvm_two_phase_dam_break.

Main function

using Tesserae
using Tesserae.Stencil
using LinearAlgebra
using Krylov: CgWorkspace, cg!
using SparseArrays: nonzeros
using SparseMatricesCSR: SparseMatrixCSR
using LinearOperators: LinearOperator
using AMGCLWrap: AMGPrecon
import Tesserae: VOF

function main()

    # Simulation parameters
    h = 0.02
    t_stop = 7.0
    width, height = 3.22, 2.5
    water_width, water_height = 1.2, 0.6
    gravity = (0.0, -9.81)
    maximum_step = 0.002
    minimum_step = 1e-10
    cfl = 0.25
    fps = 30
    output = joinpath("output", "fvm_two_phase_dam_break")
    divergence_tolerance = 1e-10

    # Material constants
    water, air = 1000.0, 1.2
    μwater, μair = 1.01e-3, 1.8e-5
    contrast, half = water-air, 0.5

Grid and initial condition

Store volume fraction and pressure at cell centers and velocity component i on faces normal to axis i. The initial fraction is the overlap of each cell with the water rectangle. Compute the overlap in grid units so that aligned cells are exactly full or empty. One halo layer supplies samples for reconstruction at the boundaries. In the stencils below, a half-cell shift such as C+ed/2 selects each cell's positive face. Shifting an interior face by ±ed/2 selects its two adjacent cells.

    # Cell and face properties
    CellProp = @NamedTuple begin
        x      :: Vec{2, Float64}
        c      :: Float64
        c_prev :: Float64 # Fraction before the current directional sweep.
        ρ      :: Float64
        μ      :: Float64
        τxx    :: Float64
        τyy    :: Float64
        H      :: Bool    # WY indicator frozen for the full time step.
        p      :: Float64
        S      :: Float64
    end
    FaceProp = @NamedTuple begin
        x  :: Vec{2, Float64}
        v  :: Float64
        vⁿ :: Float64
        m  :: Float64
        s  :: Float64
        c  :: Float64
        Fw :: Float64
        Fm :: Float64
        α  :: Float64
    end
    # Background grid
    mesh = CartesianMesh(h, (0, width), (0, height))
    cells = generate_grid(Cell(), CellProp, mesh; halo=1)
    faces = [generate_grid(Face(d), FaceProp, mesh; halo=1) for d in 1:2]
    C = Region(cells)
    water_cells = (water_width/h, water_height/h)
    for I in CartesianIndices(size(C))
        cells.c[I] = clamp(water_cells[1]-(I[1]-1), 0, 1)*clamp(water_cells[2]-(I[2]-1), 0, 1)
    end

Time integration

    e = unitoffsets(C)
    ex, ey = e
    face_regions = Region.(faces)
    interior_faces = interior.(face_regions)
    bottom, top = boundaries(face_regions[2], 2)
    solid_faces = [collect(boundaries(face_regions[1], 1)), [bottom]]
    active_faces = [interior_faces[1], intersect(face_regions[2], face_regions[2]+ey)]
    face_indices = [CartesianIndices(size(F)) for F in face_regions]
    directions = (CartesianIndex(1, 0), CartesianIndex(0, 1))
    upper_cells = top+ey/2
    solid_planes = (boundaries(mesh, 1)..., first(boundaries(mesh, 2)))
    top_plane = last(boundaries(mesh, 2))
    upper_face_rows = [halos(face, top_plane) for face in faces]

    # Store the cell-centered flux above the open top in a halo row.
    V = Region(Vertex(), C)
    vertex_flux_region = intersect(interior(V, 1), V+ey)
    cell_flux = zeros(C; halo=1)
    vertex_flux, shear = zeros(size(V)), zeros(size(V))

    # Pressure work arrays. Solid-wall gradients stay zero.
    alpha = [view(face.α, F) for (face, F) in zip(faces, face_regions)]
    gradients = [zeros(size(F)) for F in face_regions]
    trial_pressure = zeros(size(C))
    div = similar(trial_pressure)
    div_vector = vec(div)
    rhs = similar(div_vector)
    pressure = view(cells.p, C)
    A = create_sparse_matrix(C, C, (0ex, -ex, ex, -ey, ey); ndofs=1)
    amg_matrix = SparseMatrixCSR{0}(transpose(A))
    solver = CgWorkspace(length(pressure), length(pressure), Vector{Float64})
    operator = LinearOperator(Float64, length(pressure), length(pressure), true, true, (y, x) -> begin
        copyto!(vec(trial_pressure), x)
        pressure_product!(div, trial_pressure, gradients, alpha, C, interior_faces, e, top, h)
        copyto!(y, div_vector)
    end)

    # Output
    pvdfile = joinpath(mkpath(output), "paraview")
    closepvd(openpvd(pvdfile))
    savepoints = LinRange(0, t_stop, max(2, ceil(Int, t_stop*fps)+1))
    next_save = 1
    acceleration = sum(abs, gravity)/h
    viscous_step = h^2/(8max(μwater/water, μair/air))
    time, step = 0.0, 0
    while time < t_stop-eps(t_stop)

Time step

The CFL bound includes the advective rate and gravitational acceleration. With $R=\sum_d\max|u_d|/h$ and $A=\sum_d|g_d|/h$, the step satisfies $R\Delta t+A\Delta t^2\leq\mathrm{CFL}$. The explicit viscous update sets a second upper bound.

        rate = sum(maximum(abs, view(face.v, F))/h for (face, F) in zip(faces, face_regions))
        dt = min(maximum_step, viscous_step, 2cfl/(rate+sqrt(rate^2+4acceleration*cfl)))
        dt > minimum_step || error("time step collapsed at t=$time")
        remaining = t_stop-time
        # Split the remainder if taking a full step would leave a tiny final step.
        if 0 < remaining-dt ≤ minimum_step
            dt = remaining/2
        end
        dt = min(remaining, dt)

Volume and momentum transport

Freeze the indicator $H^n=\mathbf{1}_{c^n>1/2}$ and the face velocity for the full time step. For one directional sweep, write $\lambda_f=u^n_{d,f}\Delta t/h$ and let $F^w_f$ be the signed PLIC water flux divided by cell volume. The cell update is

\[c_K^{\rm new}=c_K^{\rm old}+F^w_--F^w_++H^n_K(\lambda_+-\lambda_-).\]

The mass flux is $F^m=\rho_a\lambda+(\rho_w-\rho_a)F^w$. A staggered velocity control volume spans two half cells. For each of its faces, select the upwind velocity for each half-face mass flux independently.

\[\mathcal{F}_i=\tfrac12(F^m_1 u_{i,\mathrm{up},1}+F^m_2 u_{i,\mathrm{up},2}).\]

The split mass source is $S=(\rho_a+(\rho_w-\rho_a)H^n)(\lambda_+-\lambda_-)$. Average it and the density onto velocity faces. At walls, reflect the fraction and velocity, reversing the normal velocity component. Extrapolate both advecting velocity components and their mass fluxes across the open top for the staggered momentum balance.

        @stencil begin
            cells.ρ[C] = air+contrast*cells.c[C]
            cells.ρ[upper_cells] = cells.ρ[upper_cells-ey]
            cells.H[C] = cells.c[C] > half
        end
        for (i, F) in enumerate(face_regions)
            face, ei = faces[i], e[i]
            I, walls = active_faces[i], solid_faces[i]
            H = upper_face_rows[i]
            @stencil begin
                face.vⁿ[F] = face.v[F]
                face.vⁿ[H] = face.vⁿ[H-ey]
                face.m[I] = (cells.ρ[I-ei/2]+cells.ρ[I+ei/2])*half*face.vⁿ[I]
                face.m[walls] = 0
            end
        end

        # Sweep in each direction, alternating the starting axis.
        for sweep in 0:1
            d = mod(step+sweep, 2)+1
            sweepface, ed, factor = faces[d], e[d], dt/h
            Fd, Id = face_regions[d], interior_faces[d]
            walls = solid_faces[d]
            H = upper_face_rows[d]
            @stencil cells.c_prev[C] = cells.c[C]
            boundary_halos!(cells.c_prev, cells, solid_planes, top_plane, ey)
            @stencil begin
                sweepface.Fw[Id] = upstream_flux(factor*sweepface.vⁿ[Id], (cells.c_prev,),
                    face_indices[d][Id], directions[d], d)
                sweepface.Fw[walls] = 0
            end
            if d == 2
                # Air crosses the atmospheric boundary; water stays below it.
                maximum(abs, view(cells.c_prev, top-ey/2)) ≤ 1e-10 ||
                    error("water reached the atmospheric boundary at t=$time")
                @stencil sweepface.Fw[top] = 0
            end
            @stencil begin
                cells.c[C] += sweepface.Fw[C-ed/2]-sweepface.Fw[C+ed/2]+
                    cells.H[C]*(factor*sweepface.vⁿ[C+ed/2]-factor*sweepface.vⁿ[C-ed/2])
                sweepface.Fm[Fd] = air*factor*sweepface.vⁿ[Fd]+contrast*sweepface.Fw[Fd]
                sweepface.Fm[H] = sweepface.Fm[H-ey]
                cells.S[C] = (air+contrast*cells.H[C])*(factor*sweepface.vⁿ[C+ed/2]-factor*sweepface.vⁿ[C-ed/2])
                cells.S[upper_cells] = cells.S[upper_cells-ey]
                cells.ρ[C] = air+contrast*cells.c[C]
                cells.ρ[upper_cells] = cells.ρ[upper_cells-ey]
            end
            for i in 1:2
                face, ei = faces[i], e[i]
                Fi, Ii = face_regions[i], active_faces[i]
                flux = i == d ? cell_flux : vertex_flux
                flux_regions = i != d ? (vertex_flux_region,) : d == 2 ? (C, upper_cells) : (C,)
                @stencil face.c[Fi] = (cells.c_prev[Fi-ei/2]+cells.c_prev[Fi+ei/2])*half
                boundary_halos!(face.c, face, solid_planes, top_plane, ey)
                boundary_halos!(face.v, face, solid_planes, top_plane, ey, i)
                @stencil face.s[Fi] = limited_slope(face.v[Fi-ed], face.v[Fi], face.v[Fi+ed],
                    face.c[Fi-ed], face.c[Fi], face.c[Fi+ed])
                fillzero!(flux)
                for Q in flux_regions
                    @stencil flux[Q] = half*(
                        sweepface.Fm[Q-ei/2]*donor_value(factor*sweepface.vⁿ[Q-ei/2],
                            face.v[Q-ed/2], face.v[Q+ed/2], face.s[Q-ed/2], face.s[Q+ed/2]) +
                        sweepface.Fm[Q+ei/2]*donor_value(factor*sweepface.vⁿ[Q+ei/2],
                            face.v[Q-ed/2], face.v[Q+ed/2], face.s[Q-ed/2], face.s[Q+ed/2]))
                end
                @stencil begin
                    face.m[Ii] += flux[Ii-ed/2]-flux[Ii+ed/2]+
                        face.vⁿ[Ii]*(cells.S[Ii-ei/2]+cells.S[Ii+ei/2])*half
                    face.v[Ii] = face.m[Ii]/((cells.ρ[Ii-ei/2]+cells.ρ[Ii+ei/2])*half)
                end
            end
        end

Viscosity and gravity

Normal stresses are cell-centered and shear stress is stored at vertices. Add their divergence and gravity to the transported velocity to obtain $u^\star$.

        @stencil cells.μ[C] = μair+(μwater-μair)*cells.c[C]
        boundary_halos!(cells.μ, cells, solid_planes, top_plane, ey)
        for i in 1:2
            boundary_halos!(faces[i].v, faces[i], solid_planes, top_plane, ey, i)
        end
        u, v = faces[1].v, faces[2].v
        @stencil begin
            cells.τxx[C] = 2cells.μ[C]*(u[C+ex/2]-u[C-ex/2])/h
            cells.τyy[C] = 2cells.μ[C]*(v[C+ey/2]-v[C-ey/2])/h
            cells.τyy[upper_cells] = 0
            shear[V] = (cells.μ[V-ex/2-ey/2]+cells.μ[V+ex/2-ey/2]+
                cells.μ[V-ex/2+ey/2]+cells.μ[V+ex/2+ey/2])/4*
                ((u[V+ey/2]-u[V-ey/2])/h+(v[V+ex/2]-v[V-ex/2])/h)
        end
        fillzero!(div)
        for (i, I) in enumerate(active_faces)
            face, ei, et = faces[i], e[i], e[3-i]
            τ = i == 1 ? cells.τxx : cells.τyy
            walls = solid_faces[i]
            @stencil face.α[I] = 2/(cells.ρ[I-ei/2]+cells.ρ[I+ei/2])
            if i == 2
                @stencil face.α[top] = 1/cells.ρ[top-ey/2]
            end
            @stencil begin
                face.v[I] += dt*(gravity[i]+face.α[I]*(
                    (τ[I+ei/2]-τ[I-ei/2])/h+(shear[I+et/2]-shear[I-et/2])/h))
                face.v[walls] = 0
                div[C] += (face.v[C+ei/2]-face.v[C-ei/2])/h
            end
        end

Pressure projection

Solve $-D(\alpha Gp)=-Du^\star/\Delta t$ and correct $u^{n+1}=u^\star-\Delta t\,\alpha Gp$. At the open top, atmospheric pressure is imposed at a half-cell distance. The sparse matrix is used for AMG, while CG applies the same pressure gradient used in the velocity correction. Verify the actual divergence independently of CG's preconditioned norm.

        @. rhs = -div_vector/dt
        pressure_tolerance = divergence_tolerance/(10dt*sqrt(length(pressure)))
        assemble_pressure!(amg_matrix, alpha, size(C), h, directions)
        preconditioner = AMGPrecon(amg_matrix)
        try
            cg!(solver, operator, rhs, vec(pressure); M=preconditioner, ldiv=true,
                atol=pressure_tolerance, rtol=0.0)
        finally
            finalize(preconditioner)
        end
        solver.stats.solved || error("pressure CG did not converge at t=$time: $(solver.stats.status)")
        copyto!(vec(pressure), solver.x)
        pressure_gradient!(gradients, pressure, interior_faces, e, top, h)
        fillzero!(div)
        for (i, I) in enumerate(active_faces)
            face, ei, ∇p = faces[i], e[i], gradients[i]
            @stencil begin
                face.v[I] -= dt*face.α[I]*∇p[I]
                div[C] += (face.v[C+ei/2]-face.v[C-ei/2])/h
            end
        end
        max_divergence = maximum(abs, div)
        max_divergence ≤ divergence_tolerance || error("projection divergence $max_divergence exceeds $divergence_tolerance")
        time += dt
        step += 1

        # Write results
        if next_save ≤ length(savepoints) && time+eps(t_stop) ≥ savepoints[next_save]
            save_time = time
            filename = string(pvdfile, step)
            openpvd(pvdfile; append=true) do pvd
                openvtk(filename, mesh) do vtk
                    vtk["Water fraction"] = cells.c[C]
                    vtk["Pressure"] = cells.p[C]
                    pvd[save_time] = vtk
                end
            end
            next_save = searchsortedlast(savepoints, time+eps(t_stop))+1
        end
    end
end

Pressure operator and preconditioner

Each interior face contributes $\alpha_f/h^2$ to both diagonal entries and its negative to both off-diagonal entries. The open-top pressure condition adds only a diagonal term, with twice that weight.

function assemble_pressure!(A, alpha, size_c, h, directions)
    fillzero!(nonzeros(A))
    ids = LinearIndices(size_c)
    for I in CartesianIndices(size_c), d in 1:2
        if I[d] == size_c[d]
            if d == 2
                A[ids[I], ids[I]] += 2alpha[d][I+directions[d]]/h^2
            end
            continue
        end
        J = I+directions[d]
        i, j = ids[I], ids[J]
        w = alpha[d][J]/h^2
        A[i, i] += w
        A[i, j] -= w
        A[j, i] -= w
        A[j, j] += w
    end
end

function pressure_gradient!(gradients, p, interior_faces, e, top, h)
    for (I, ed, ∇p) in zip(interior_faces, e, gradients)
        @stencil ∇p[I] = (p[I+ed/2]-p[I-ed/2])/h
    end
    ∇py, ey = gradients[2], e[2]
    @stencil ∇py[top] = -2p[top-ey/2]/h
end

function pressure_product!(out, p, gradients, alpha, C, interior_faces, e, top, h)
    pressure_gradient!(gradients, p, interior_faces, e, top, h)
    fillzero!(out)
    for (ed, ∇p, α) in zip(e, gradients, alpha)
        @stencil out[C] -= (α[C+ed/2]*∇p[C+ed/2]-α[C-ed/2]*∇p[C-ed/2])/h
    end
    out
end

function boundary_halos!(field, grid, solid_planes, top, ey, component=0)
    for (i, wall) in enumerate(solid_planes)
        H = halos(grid, wall)
        K = reflect(H, wall)
        normal_axis = i ≤ 2 ? 1 : 2
        sign = component == normal_axis ? -1 : 1
        @stencil field[H] = sign*field[K]
    end
    H = halos(grid, top)
    @stencil field[H] = field[H-ey]
end

Local geometry and transport kernels

The integer face index labels the cell on its positive side. upstream_flux selects the donor cell. VOF.flux reconstructs the MYC plane for a mixed donor and integrates the swept water volume. The momentum flux uses a monotonized-central slope in pure-phase stencils and zero slope when the stencil crosses the interface.

@inline function upstream_flux(a, fraction, I, direction, axis)
    iszero(a) && return zero(a)
    J = a > 0 ? I-direction : I
    VOF.flux(a, fraction, J, axis)
end

function limited_slope(left, center, right, c_left, c_center, c_right)
    pure = max(c_left, c_center, c_right) ≤ 0 || min(c_left, c_center, c_right) ≥ 1
    a, b = center-left, right-center
    !pure || a*b ≤ 0 ? zero(center) : copysign(min(2abs(a), 2abs(b), abs(a+b)/2), a)
end

function donor_value(a, left, right, left_slope, right_slope)
    iszero(a) && return zero(left)
    distance = (one(a)-abs(a))/2
    a > 0 ? left+distance*left_slope : right-distance*right_slope
end
